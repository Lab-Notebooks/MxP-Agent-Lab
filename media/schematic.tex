% The Lab Notebook (the repository) holds the Computational Laboratory: staged
% automations that run in the container, where a scientist starts each stage,
% an agent works it, and the scientist signs off the gate before moving on.
% Build: latexmk -pdf schematic.tex
\documentclass[tikz, border=4pt]{standalone}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds, shapes.misc, shapes.symbols, calc}

\tikzset{
  scientist/.pic={
    \fill (0,0) circle (3.6pt);
    \draw[line width=2.6pt, line cap=round]
      (-5pt,-12pt) to[out=90,in=205] (0,-5pt) to[out=-25,in=90] (5pt,-12pt);
  },
  agent/.pic={
    \draw[fill=white, rounded corners=1.5pt] (-5pt,-4pt) rectangle (5pt,4pt);
    \fill (-2pt,0.5pt) circle (1pt) (2pt,0.5pt) circle (1pt);
    \draw (0,4pt) -- (0,6.5pt); \fill (0,7pt) circle (1pt);
  },
  gate/.pic={
    \draw[fill=white, line width=0.9pt] (0,0) circle (6pt);
    \draw[line width=1.2pt, green!45!black] (-3pt,0) -- (-0.8pt,-2.4pt) -- (3pt,2.6pt);
  },
}

\pgfdeclarelayer{paper}
\pgfsetlayers{paper,background,main}

\begin{document}
\begin{tikzpicture}[
    font=\footnotesize,
    >={Latex[length=1.8mm, width=1.4mm]},
    every path/.style={line width=0.6pt},
    box/.style={draw, rounded corners=2pt, align=center, inner sep=4pt, fill=white},
    stage/.style={box, line width=0.9pt, text width=2.3cm, minimum height=1.25cm},
    page/.style={draw, fill=white, align=left, inner sep=5pt, chamfered rectangle,
      chamfered rectangle corners=north east, chamfered rectangle xsep=5pt},
    lbl/.style={font=\scriptsize, text=black!60, align=center},
  ]

  % Stages, each worked by an agent, and the signed-off gates between them
  \foreach \n/\x/\t/\d in {rap/0/raptorize/{precision modes\\+ EVOLVE-BLOCK seed},
                          ae/3.6/alphaevolve/{evolve the seed\\under a trusted score},
                          ver/7.2/verify/{tests, checks,\\analysis}} {
    \node[stage] (\n) at (\x,0) {\textbf{\t}\\[1pt]{\scriptsize\d}};
    \path ([xshift=-10pt]\n.north east) pic{agent};
  }
  \foreach \g/\x in {g1/1.8, g2/5.4, g3/9.0}
    \coordinate (\g) at (\x,0);
  \draw[->] (rap) -- ([xshift=-6pt]g1); \draw[->] ([xshift=6pt]g1) -- (ae);
  \draw[->] (ae) -- ([xshift=-6pt]g2);  \draw[->] ([xshift=6pt]g2) -- (ver);
  \draw[->] (ver) -- ([xshift=-6pt]g3);
  \foreach \g in {g1, g2, g3} \path (\g) pic{gate};
  \node[lbl, below=7pt of g3] {merge};
  \draw[->, rounded corners=3pt, black!60] (ver.south) -- ++(0,-0.35)
    -| node[lbl, pos=0.25, below] {new constraints} (ae.south);

  % Computational Laboratory: the container running the staged automations
  \coordinate (lablow) at ([yshift=-2cm]ae.south);
  \begin{scope}[on background layer]
    \node[draw=black!50, dashed, rounded corners=3pt, fill=green!9, inner sep=8pt,
          fit=(rap) (ver) (g3) (lablow)] (lab) {};
  \end{scope}
  \node[lbl, anchor=south west, align=left] at ([shift={(3pt,2pt)}]lab.south west)
    {{\small\bfseries\color{black}Computational Laboratory}\ \
     container (\ttfamily Dockerfile\normalfont): toolchain, RAPTOR, \ttfamily ae\normalfont\\[2pt]
     each stage: ask $\to$ record $\to$ work $\to$ checkpoint $\to$ gate $\to$ hand off\\
     {\ttfamily\color{black}dev/workflows/} (system.md, plan.md) $\cdot$
     {\ttfamily\color{black}dev/tools/} (RAPTOR, evolve.py, verify.py)};

  % Pages of the notebook the lab writes to
  \node[page, fill=black!4, text width=4.1cm, below=0.9cm of lab.south -| rap.east, anchor=north]
    (app) {\textbf{\ttfamily app/}\ \ {\scriptsize the code under study}\\[1pt]
           {\ttfamily\scriptsize src/ $\cdot$ tests/ $\cdot$ params/ $\cdot$ tools/}};
  \node[page, fill=blue!7, text width=4.6cm, anchor=north] (exp) at (app.north -| ver.east)
    {\textbf{\ttfamily experiments/<yyyy-mm-dd>/}\\[2pt]
     \ttfamily\scriptsize
     \begin{tabular}{@{}l@{\quad}l@{}}
       decisions.md & evaluator.py\\ raptorize.patch & verification.yaml\\
       kernel.yaml & analysis.md\\ experiment.yaml & results/
     \end{tabular}};
  \draw[->] (lab.south -| app.north) -- node[lbl, right] {patch, build,\\run \& test} (app.north);
  \draw[->] (lab.south -| exp.north) -- node[lbl, right] {records\\every step} (exp.north);

  % Lab Notebook: the whole repository, spiral-bound
  \coordinate (nbtop) at ([yshift=0.3cm]lab.north);
  \coordinate (nblow) at ([yshift=-0.55cm]exp.south);
  \begin{pgfonlayer}{paper}
    \node[draw, rounded corners=4pt, fill=yellow!7, inner xsep=14pt, inner ysep=8pt,
          fit=(lab) (app) (exp) (nbtop) (nblow)] (nb) {};
    \draw[red!40] ([xshift=10pt]nb.north west) -- ([xshift=10pt]nb.south west);
  \end{pgfonlayer}
  \foreach \i in {1,...,13} {
    \coordinate (s) at ($(nb.north west)!{\i/14}!(nb.south west)$);
    \draw[black!55, line width=1pt] ([xshift=-4pt]s) arc[start angle=180, end angle=0, radius=4pt];
    \fill[black!55] ([xshift=4pt]s) circle (1pt);
  }
  \node[lbl, anchor=south west] at ([shift={(16pt,5pt)}]nb.south west)
    {{\small\bfseries\color{black}Lab Notebook}\ \ the repository:
     {\ttfamily AGENTS.md $\cdot$ Dockerfile $\cdot$ Makefile $\cdot$ dev/ $\cdot$ app/ $\cdot$ experiments/}};

  % AlphaEvolve service, outside the notebook
  \node[cloud, draw, dashed, cloud puffs=10, aspect=2.4, inner sep=-2pt, align=center, fill=white]
    (gcp) at ([xshift=0.45cm]ae.north |- 0,2.75) {\textbf{AlphaEvolve}\\{\scriptsize GCP, \ttfamily ae}};
  \draw[<->] (gcp.south) -- ([xshift=0.45cm]ae.north);
  \node[lbl, inner sep=1pt, anchor=north west] at ([xshift=0.5cm]ae.north |- gcp.south)
    {candidates, scores};

  % Scientist: starts each stage, answers its questions, signs off its gate
  \coordinate (sci) at (-1.6,4.5);
  \path (sci) pic{scientist};
  \node[above=2pt of sci, font=\ttfamily\scriptsize] {scientist};
  \draw[black!70, line width=0.9pt] ([xshift=0.35cm, yshift=-0.2cm]sci) -- (9.0,4.3) coordinate (railend);
  \node[lbl, anchor=south east] at (railend) {decides; agents propose, execute, report};
  \foreach \n in {rap, ae, ver}
    \draw[<->, black!70] ([xshift=-0.75cm]\n.north |- 0,4.3) -- ([xshift=-0.75cm]\n.north);
  \foreach \g in {g1, g2, g3}
    \draw[<-, green!45!black] ([yshift=6pt]\g) -- (\g |- 0,4.3);
  \node[lbl, fill=white, inner sep=1pt, align=left, anchor=west] at ([xshift=-0.72cm]rap.north |- 0,3.75)
    {start,\\checkpoints};
  \node[lbl, fill=white, inner sep=1pt, text=green!40!black, anchor=west] at (1.85,3.75) {sign off};
\end{tikzpicture}
